extends "res://scripts/tests/kart_drive_feedback_regression.gd"

func _release_audio(playbacks: Array[WeakRef]) -> bool:
	var held: RefCounted
	for playback: WeakRef in playbacks:
		held = playback.get_ref()
		if held != null:
			break
	if held == null:
		push_error("Negative fixture could not retain a real active playback")
		return false
	var started := Time.get_ticks_msec()
	var released: bool = await super._release_audio(playbacks)
	print("NEGATIVE_HELD_PLAYBACK released=%s elapsed_ms=%d" % [released, Time.get_ticks_msec() - started])
	held = null
	# Release the deliberately retained fixture before process exit; the
	# original timeout result still propagates into the real regression.
	await super._release_audio(playbacks)
	return released
